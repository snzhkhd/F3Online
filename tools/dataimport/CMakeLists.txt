cmake_minimum_required(VERSION 3.10)
if(NOT PROJECT_NAME)
  project(dataimport CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release)
  endif()
endif()
# Reuses the strict parsers of tools/mapinfo by including its main.cpp (main renamed).
add_executable(f3_dataimport main.cpp)
target_include_directories(f3_dataimport PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../mapinfo)
if(MSVC)
  target_compile_options(f3_dataimport PRIVATE /W3)
  target_compile_definitions(f3_dataimport PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
